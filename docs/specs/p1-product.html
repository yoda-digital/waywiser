    SttResult(String),
    SttPartial(String),
    AgentStreamChunk(String),
    AgentStreamDone,
    TtsChunkReady(AudioChunk),
    TtsPlaybackDone,
    BargeIn,                // VAD fires during Speaking
    Cancel,
    ErrorOccurred(VoiceError),
}

// Transitions
// Idle → Listening:           on MicActivated
// Listening → Recognizing:    on VadSpeechEnd
// Recognizing → Processing:   on SttResult (feed to Pi agent)
// Processing → Speaking:      on AgentStreamChunk (first chunk)
// Speaking → Speaking:        on AgentStreamChunk (subsequent)
// Speaking → Idle:            on TtsPlaybackDone
// Speaking → BargeInDetected: on BargeIn
// BargeInDetected → Listening: immediate (stop TTS, steer Pi)
// Any → Idle:                 on Cancel
// Any → Error:                on ErrorOccurred

4. CaptureSystem Kotlin

/// Unified entry point for all input modalities
class CaptureManager(
    private val runtime: WaywiserRuntime,
    private val voiceManager: VoiceManager,
) {
    sealed interface CaptureSource {
        data class Text(val content: String) : CaptureSource
        data class Voice(val transcript: String, val audio: Uri?) : CaptureSource
        data class Camera(val imageUri: Uri, val ocr: String?) : CaptureSource
        data class ShareSheet(val intent: Intent) : CaptureSource
        data class File(val uri: Uri, val mimeType: String) : CaptureSource
    }

    /// All captures → Observation with appropriate RetentionClass
    suspend fun capture(source: CaptureSource): Observation {
        val obs = when (source) {
            is CaptureSource.Text -> Observation(
                kind = ObservationKind.UserInput,
                value = source.content,
                retention = RetentionClass.Session,   // promoted by Brain if durable
            )
            is CaptureSource.Camera -> Observation(
                kind = ObservationKind.VisualCapture,
                value = source.ocr ?: "[image]",
                retention = RetentionClass.Experience, // user explicitly captured
            )
            // ... other sources
        }
        runtime.ingestObservation(obs)
        return obs
    }
}

9. SecurityKernel — Full Implementation Rust crate: waywiser-security

This is the most critical component in the entire system. It implements blueprint invariant I2: "LLMs may propose protected state transitions. They do not authorize them."

Authorization Flow

/// The authorization pipeline — each step can only maintain or INCREASE risk
impl SecurityKernel {
    pub fn authorize(&self, intent: &ActionIntent) -> SecurityDecision {
        // Step 1: Capability exists?
        let cap = match self.registry.get(&intent.capability) {
            Some(c) => c,
            None => return SecurityDecision::Denied(
                DenialReason::UnknownCapability(intent.capability.clone())
            ),
        };

        // Step 2: Risk classification (from CapabilitySpec)
        let risk = cap.risk;

        // Step 3: Waywiser policy check
        if let Some(denial) = self.policy.check(intent, &cap) {
            return SecurityDecision::Denied(denial);
        }

        // Step 4: OS permission available?
        if let Some(os_perm) = &cap.os_permission {
            if !self.os_permissions.is_granted(os_perm) {
                return SecurityDecision::Denied(
                    DenialReason::OsPermissionMissing(os_perm.clone())
                );
            }
        }

        // Step 5: Active approval lease?
        if let Some(lease) = self.find_valid_lease(intent) {
            if lease.executions_used < lease.max_executions {
                return SecurityDecision::Allowed(AuthorizationSource::Lease(lease.id));
            }
            // Lease exhausted — fall through to approval required
        }

        // Step 6: Determine approval kind based on risk
        match risk {
            RiskLevel::ReadPersonal => SecurityDecision::Allowed(
                AuthorizationSource::DefaultPolicy
            ),
            RiskLevel::DeviceControl
            | RiskLevel::CrossAppWrite => SecurityDecision::RequiresApproval(
                ApprovalKind::UserConfirm
            ),
            RiskLevel::Communication
            | RiskLevel::Financial
            | RiskLevel::Destructive => SecurityDecision::RequiresApproval(
                ApprovalKind::BiometricConfirm
            ),
            RiskLevel::UiUnclassifiedWrite => SecurityDecision::RequiresApproval(
                ApprovalKind::UserConfirm
            ),
        }
    }
}

enum SecurityDecision {
    Allowed(AuthorizationSource),
    RequiresApproval(ApprovalKind),
    Denied(DenialReason),
}

enum AuthorizationSource {
    DefaultPolicy,
    Lease(LeaseId),
    UserApproval(DateTime<Utc>),
    BiometricApproval(DateTime<Utc>),
}

enum ApprovalKind {
    UserConfirm,        // tap "Allow"
    BiometricConfirm,   // fingerprint / face + tap

struct ApprovalLease {
    id: LeaseId,
    capability: CapabilityName,
    scope: LeaseScope,
    constraints: LeaseConstraints,
    granted_by: AuthorizationSource,
    granted_at: DateTime<Utc>,
    valid_until: DateTime<Utc>,
    max_executions: u32,
    executions_used: u32,
    revoked: bool,
}

struct LeaseScope {
    account: Option<String>,        // e.g., "work" calendar
    target: Option<String>,         // e.g., specific contact
    action_subset: Option<Vec<String>>, // e.g., ["read", "update"]
}

struct LeaseConstraints {
    forbid: Vec<String>,            // e.g., ["delete"]
    max_per_hour: Option<u32>,
    require_evidence: bool,
}

impl SecurityKernel {
    fn find_valid_lease(&self, intent: &ActionIntent) -> Option<&ApprovalLease> {
        self.leases.iter().find(|l| {
            l.capability == intent.capability
            && !l.revoked
            && l.valid_until > Utc::now()
            && l.scope.matches(intent)
            && !l.constraints.forbids(intent)
        })
    }

    fn grant_lease(&mut self, lease: ApprovalLease) -> LeaseId;
    fn revoke_lease(&mut self, id: LeaseId);
    fn record_execution(&mut self, lease_id: LeaseId) -> Result<(), DenialReason>;
}

        let inference_ok = matches!(status, InferenceStatus::Available { .. });
        Self {

            lexical_recall: true,
            cached_calendar: true,
            deterministic_automations: true,
            work_state: true,
            activity_ledger: true,
            deliberative_conversation: inference_ok,
            brain_reflection: inference_ok,
            skill_compilation: inference_ok,
            research: inference_ok, // also needs general network
            vlm_reasoning: inference_ok,
        }
    }
}

P1 Acceptance Criteria

  1. Voice end-to-end: User activates mic → speaks query → STT transcribes → Pi agent processes → streamed response → TTS plays answer. Measured latency from speech-end to first TTS audio < 3s on company network.
  2. Barge-in: During TTS playback, user speaks → TTS stops within 200ms → new utterance captured → Pi receives steer event → new response streams.
  3. Notification classification: Notification from user-designated important contact → classified as minimum NOTIFY regardless of app-declared priority. Routine CI bot notification → capped at configured ceiling.
  4. Calendar mutation flow: Voice command "Move my dentist to Thursday" → ActionIntent(calendar.update) → SecurityKernel authorizes (lease or user confirm) → calendar updated → ActionReceipt → visible in ActivityLedger with undo button.
  5. Share sheet capture: User shares text/URL/image from any app → ShareActivity receives → Observation created with RetentionClass.Experience → available in memory recall.
  6. Delegation cycle: WorkItem created → assigned to Leaf agent → agent completes within budget → result stored → status transitions to Review → user reviews → marked Done.
  7. Proactive conflict detection: Two calendar events overlap → ProactiveEngine detects via CalendarService.conflicts() → user notified with suggested resolution.
  8. APK update verification: New release on GitLab → UpdateManager fetches manifest → verifies Ed25519 signature → downloads APK → verifies SHA-256 → verifies APK signing cert → prompts user to install. Tampered manifest or APK → rejected with clear error.
  9. Offline graceful degradation: Network disconnects → UI shows degraded banner within 5s → captures continue → memory writes continue → conversation shows "offline" → network returns → DeferredQueue drains → banner clears.
  10. Background reliability: BackgroundReliabilityManager schedules probe → measures actual vs expected execution time → classifies device health → health reported in TrustCenter → if SeverelyRestricted, user guided to battery optimization settings.
  11. Security denial: Agent proposes action with unknown capability → SecurityKernel returns Denied(UnknownCapability) → no side effect occurs → AuditEntry recorded → user not disturbed.
  12. Lease lifecycle: User grants 1-hour calendar.update lease → executions succeed → lease expires → next calendar.update requires fresh approval → confirmed via AuditEntry timestamps.
  13. Daily soak test: 8 hours of mixed use (voice, text, background, notifications) → battery drain attributable to Waywiser < 15% → zero ANR → zero crash → memory stable (no leak trend in heap profiling).
  14. Process death recovery: Kill Waywiser process during active voice session → user relaunches → app recovers session from SQLite → shows last conversation state → can resume interaction without data loss.
  15. Safe mode: Simulate 3 consecutive startup crashes → app enters safe mode → disables agent runtime, Accessibility, voice, background → update check works → diagnostics accessible → user can exit safe mode.
