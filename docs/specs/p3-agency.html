    /// semantic tree may or may not still be readable.

/// A snapshot of the current accessibility tree for a window.
pub struct TreeSnapshot {
    pub window_id:      i32,
    pub package:        String,
    pub root:           A11yNode,
    pub quality:        TreeQuality,
    pub secure_state:   SecureWindowState,
    pub captured_at:    DateTime<Utc>,
}

/// Assess tree quality from node metadata density.
pub fn assess_tree_quality(root: &A11yNode) -> TreeQuality {
    let total = count_nodes(root);
    let with_id = count_with(root, |n| n.resource_id.is_some());
    let with_desc = count_with(root, |n| {
        n.content_description.is_some() || n.text.is_some()
    });
    let ratio = (with_id + with_desc) as f64 / (total.max(1) * 2) as f64;
    match ratio {
        r if r >= 0.6 => TreeQuality::Good,
        r if r >= 0.3 => TreeQuality::Partial,
        r if r >= 0.1 => TreeQuality::Poor,
        _              => TreeQuality::Unusable,
    }
}

        return node.performAction(action.toAndroidAction())
    }

2. RiskClassifier Rust · waywiser-security ext.

Blueprint §§23–23.5. Five-layer deterministic pipeline. Invariant: risk may only stay the same or increase as layers execute.

/// Five-layer risk classification pipeline.
/// Each layer can only maintain or RAISE the current risk floor.
pub struct RiskClassifier {
    sensitive_packages:    HashMap<String, PackagePolicy>,
    automation_profiles:   Vec<AutomationProfile>,
    semantic_rules:        Vec<SemanticRule>,
    primitive_floors:      HashMap<PrimitiveActionKind, RiskLevel>,
}

/// Per-package security policy (Layer 1).
pub struct PackagePolicy {
    pub package:    String,
    pub automation: AutomationPolicy,  // Allow | Block
    pub floor:      Option<RiskLevel>,
}

/// Semantic text rule (Layer 3).
pub struct SemanticRule {
    pub pattern:  regex::Regex,        // e.g. r"send|reply|publish|post"
    pub floor:    RiskLevel,           // e.g. Communication
}

/// Input to the classifier.
pub struct ClassificationRequest {
    pub package:      String,
    pub node:         A11yNode,
    pub action:       PrimitiveActionKind,
    pub llm_hint:     Option<LlmRiskHint>,  // non-authoritative, logged only
}

/// Output from the classifier.
pub struct ClassificationResult {
    pub final_risk:     RiskLevel,
    pub layer_trace:    Vec<LayerDecision>,  // audit trail
    pub llm_hint_used:  bool,               // always false for final_risk
    pub approval_mode:  ApprovalMode,       // Auto | AskUser | Block
}

/// One layer's contribution to the trace.
pub struct LayerDecision {
    pub layer:     ClassificationLayer,
    pub matched:   bool,
    pub risk_at:   RiskLevel,   // risk AFTER this layer (≥ previous)
    pub reason:    String,
}

#[derive(Copy, Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum ClassificationLayer {
    SensitivePackage,       // Layer 1
    ReviewedProfile,        // Layer 2
    SemanticRule,           // Layer 3
    PrimitiveActionFloor,   // Layer 4
    UnknownFallback,        // Layer 5
}

#[derive(Copy, Clone, Debug, PartialEq, Eq)]
pub enum PrimitiveActionKind {
    InspectTree,
    Scroll,
    OpenApp,
    Click,
    Toggle,
    TypeText,
    Paste,
    Gesture,
}

impl RiskClassifier {
    /// Classify risk. Layers execute in order 1→5.
    /// Risk floor can only increase; decrease is a logic error.
    pub fn classify(&self, req: &ClassificationRequest) -> ClassificationResult {
        let mut risk = RiskLevel::None;
        let mut trace = Vec::new();

        // Layer 1: Sensitive-package policy
        if let Some(policy) = self.sensitive_packages.get(&req.package) {
            if policy.automation == AutomationPolicy::Block {
                return ClassificationResult::blocked(
                    &trace, "sensitive-package: automation blocked"
                );
            }
            if let Some(floor) = policy.floor {
                risk = risk.max(floor);
            }
            trace.push(LayerDecision::hit(ClassificationLayer::SensitivePackage, risk));
        }

        // Layer 2: Reviewed automation profile
        if let Some(profile) = self.find_profile(&req.package, &req.node) {
            risk = risk.max(profile.risk);
            trace.push(LayerDecision::hit(ClassificationLayer::ReviewedProfile, risk));
        }

        // Layer 3: Deterministic semantic rules
        let text = req.node.combined_text();
        for rule in &self.semantic_rules {
            if rule.pattern.is_match(&text) {
                risk = risk.max(rule.floor);
                trace.push(LayerDecision::hit(ClassificationLayer::SemanticRule, risk));
                break;
            }
        }

        // Layer 4: Primitive action-type floor
        if let Some(&floor) = self.primitive_floors.get(&req.action) {
            risk = risk.max(floor);
            trace.push(LayerDecision::hit(ClassificationLayer::PrimitiveActionFloor, risk));
        }

        // Layer 5: Unknown fallback
        if risk == RiskLevel::None {
            risk = RiskLevel::UiUnclassifiedWrite;
            trace.push(LayerDecision::hit(ClassificationLayer::UnknownFallback, risk));
        }

        // Log LLM hint but NEVER let it lower risk
        if let Some(hint) = &req.llm_hint {
            tracing::info!(llm_hint = ?hint, final_risk = ?risk,
                "LLM risk hint logged (non-authoritative)");
        }

        ClassificationResult {
            final_risk: risk,
            layer_trace: trace,
            llm_hint_used: false,
            approval_mode: risk.default_approval_mode(),
        }
    }
}

4. TOCTOUProtection Kotlin + Rust

Blueprint §24. The UI changes between plan and execution — fingerprint-verify immediately before acting.

/// Fingerprint of a UI node at planning time.
/// Used to verify the target hasn't changed before executing.
#[derive(Clone, Debug)]
pub struct NodeFingerprint {
    pub package:              String,
    pub window_id:            i32,
    pub resource_id:          Option<String>,
    pub class_name:           String,
    pub role:                 Option<String>,
    pub normalized_text:      Option<String>,
    pub content_description:  Option<String>,
    pub ancestor_signature:   Vec<String>,  // class names up the tree
    pub state:                Option<String>,
    pub approximate_bounds:   Rect,
}

/// Result of fingerprint comparison.
#[derive(Debug)]
pub enum FingerprintMatch {
    Exact,
    Partial { mismatches: Vec<FingerprintMismatch> },
    NoMatch,
}

#[derive(Debug)]
pub enum FingerprintMismatch {
    TextChanged { expected: String, actual: String },
    BoundsMoved { delta_x: i32, delta_y: i32 },
    StateChanged { expected: String, actual: String },
    AncestorChanged,
    ResourceIdMissing,
}

/// Policy: what FingerprintMatch is acceptable per risk level?
pub fn acceptable_match(risk: RiskLevel, m: &FingerprintMatch) -> bool {
    match (risk, m) {
        (_, FingerprintMatch::Exact) => true,
        (_, FingerprintMatch::NoMatch) => false,
        // High risk: require Exact — no partial allowed
        (RiskLevel::Financial | RiskLevel::Destructive | RiskLevel::Communication,
         FingerprintMatch::Partial { .. }) => false,
        // Lower risk: partial OK if mismatches are cosmetic
        (_, FingerprintMatch::Partial { mismatches }) =>
            mismatches.iter().all(|m| m.is_cosmetic()),
    }
}

/// Pre-action verification: re-resolve, fingerprint, decide.
pub fn verify_before_action(
    bridge: &dyn A11yBridge,
    planned: &NodeFingerprint,
    risk: RiskLevel,
) -> Result<A11yNode, TocTouError> {
    let snapshot = bridge.capture_tree_snapshot(planned.window_id)?;
    let resolved = find_by_fingerprint(&snapshot.root, planned)
        .ok_or(TocTouError::TargetNotFound)?;
    let current_fp = NodeFingerprint::from_node(&resolved);
    let comparison = planned.compare(&current_fp);
    if acceptable_match(risk, &comparison) {
        Ok(resolved)
    } else {
        Err(TocTouError::FingerprintMismatch {
            planned: planned.clone(),
            actual: current_fp,
            comparison,
        })
    }
}

#[derive(Debug)]
pub enum TocTouError {
    TargetNotFound,
    FingerprintMismatch {
        planned:    NodeFingerprint,
        actual:     NodeFingerprint,
        comparison: FingerprintMatch,
    },
    TreeUnavailable,
    SecureWindow,
}

/// Expected outcome of a UI action, defined at planning time.
#[derive(Clone, Debug)]
pub struct ExpectedTransition {
    /// What the current window/activity should look like after the action.
    pub expected_package:    Option<String>,
    pub expected_window:     Option<String>,
    /// Nodes that should appear (e.g. toast, confirmation).
    pub expected_indicators: Vec<TransitionIndicator>,
    /// Nodes that should disappear (e.g. the dialog we just confirmed).
    pub expected_removals:   Vec<String>,  // resource_ids
    /// Max time to wait for transition before declaring Unknown.
    pub verification_timeout: Duration,
}

pub struct TransitionIndicator {
    pub kind:     IndicatorKind,
    pub text:     Option<String>,
}

pub enum IndicatorKind {
    Toast,
    Dialog,
    ActivityChange,
    ViewAppeared,
    ViewDisappeared,
}

/// Post-action verification. Called after every significant UI action.
pub fn verify_after_action(
    bridge: &dyn A11yBridge,
    expected: &ExpectedTransition,
) -> VerificationResult {
    // Wait briefly for transition to settle
    std::thread::sleep(expected.verification_timeout.min(Duration::from_millis(500)));

    let snapshot = match bridge.capture_tree_snapshot_best_effort() {
        Ok(s)  => s,
        Err(_) => return VerificationResult::Unknown,
    };

    let indicators_present = expected.expected_indicators.iter()
        .filter(|ind| find_indicator(&snapshot.root, ind).is_some())
        .count();

    let removals_gone = expected.expected_removals.iter()
        .filter(|rid| find_by_resource_id(&snapshot.root, rid).is_none())
        .count();

    let total_checks = expected.expected_indicators.len()
        + expected.expected_removals.len();

    if total_checks == 0 {
        return VerificationResult::Unknown;
    }

    let passed = indicators_present + removals_gone;
    let ratio = passed as f64 / total_checks as f64;

    match ratio {
        r if r >= 0.8 => VerificationResult::Verified,
        r if r >= 0.4 => VerificationResult::Likely,
        _              => VerificationResult::Unexpected,
    }
}

// Verified | Likely | Unexpected | Unknown

6. CrossAppWorkflows Rust · waywiser-workflows

Multi-step action sequences across apps. Not true ACID — but explicit tracking of what succeeded and what failed, with compensating actions for reversible steps.

/// A multi-step cross-app workflow.
pub struct Workflow {
    pub id:         Uuid,
    pub name:       String,
    pub goal_id:    Option<GoalId>,
    pub steps:      Vec<WorkflowStep>,
    pub status:     WorkflowStatus,
    pub created_at: DateTime<Utc>,
    pub updated_at: DateTime<Utc>,
}

pub struct WorkflowStep {
    pub index:              usize,
    pub action_intent:      ActionIntent,
    pub pre_conditions:     Vec<PreCondition>,
    pub expected_outcome:   ExpectedTransition,
    pub actual_outcome:     Option<VerificationResult>,
    pub receipt:            Option<ActionReceipt>,
    pub status:             StepStatus,
}

#[derive(Copy, Clone, Debug)]
pub enum WorkflowStatus {
    Planning,
    Executing { current_step: usize },
    Completed,
    PartiallyCompleted { failed_at: usize },
    Aborted { reason: AbortReason },
    RollingBack { from_step: usize },
}

#[derive(Copy, Clone, Debug)]
pub enum StepStatus {
    Pending,
    Executing,
    Succeeded,
    Failed,
    Skipped,
    RolledBack,
    UnknownSideEffect,
}

pub enum AbortReason {
    VerificationUnexpected,
    TocTouMismatch,
    SecurityDenied,
    UserCancelled,
    ProcessDeath,
}

pub struct PreCondition {
    pub kind:    PreConditionKind,
    pub check:   String,  // human-readable description for audit
}

pub enum PreConditionKind {
    AppInstalled(String),
    PermissionGranted(Permission),
    NetworkAvailable,
    LeaseValid(Uuid),
    PriorStepSucceeded(usize),
}

impl Workflow {
    /// Execute the workflow step-by-step.
    /// Between each step: verify pre-conditions and post-action outcome.
    /// On UNEXPECTED: halt and record partial completion.
    pub async fn execute(
        &mut self,
        kernel: &SecurityKernel,
        bridge: &dyn A11yBridge,
    ) -> WorkflowResult {
        for step in &mut self.steps {
            // 1. Check pre-conditions
            if !step.check_pre_conditions() {
                step.status = StepStatus::Failed;
                self.status = WorkflowStatus::PartiallyCompleted {
                    failed_at: step.index
                };
                return WorkflowResult::Partial;
            }

            // 2. Run through full ActionPipeline
            let receipt = kernel.process_intent(&step.action_intent).await;

            // 3. Post-action verification
            let verification = verify_after_action(bridge, &step.expected_outcome);
            step.actual_outcome = Some(verification);
            step.receipt = Some(receipt);

            match verification {
                VerificationResult::Unexpected => {
                    step.status = StepStatus::Failed;
                    self.status = WorkflowStatus::Aborted {
                        reason: AbortReason::VerificationUnexpected
                    };
                    // HALT — do not execute further steps
                    return WorkflowResult::Aborted;
                }
                VerificationResult::Unknown => {
                    step.status = StepStatus::UnknownSideEffect;
                    // Continue with caution — next step checks pre-conditions
                }
                _ => {
                    step.status = StepStatus::Succeeded;
                }
            }
        }
        self.status = WorkflowStatus::Completed;
        WorkflowResult::Completed
    }

    /// Attempt to roll back completed reversible steps.
    pub async fn rollback(
        &mut self,
        kernel: &SecurityKernel,
    ) -> RollbackResult {
        let mut rolled = 0;
        for step in self.steps.iter_mut().rev() {
            if step.status != StepStatus::Succeeded { continue; }
            if let Some(receipt) = &step.receipt {
                if receipt.reversible && receipt.undo_token.is_some() {
                    // Execute compensating action
                    kernel.execute_undo(receipt).await;
                    step.status = StepStatus::RolledBack;
                    rolled += 1;
                }
            }
        }
        RollbackResult { steps_rolled_back: rolled }
    }
}

/// Multi-agent coordination: shared work graph conflict detection.
pub struct AgentCoordinator {
    active_agents: HashMap<AgentId, AgentState>,
    work_locks:    HashMap<WorkItemId, AgentId>,
}

impl AgentCoordinator {
    /// Claim a work item. Prevents two agents from working on the same item.
    pub fn claim_work_item(
        &mut self,
        agent: AgentId,
        item: WorkItemId,
    ) -> Result<(), ConflictError> {
        if let Some(holder) = self.work_locks.get(&item) {
            if *holder != agent {
                return Err(ConflictError::AlreadyClaimed {
                    item, holder: *holder
                });
            }
        }
        self.work_locks.insert(item, agent);
        Ok(())
    }

    /// Enforce delegation depth. Invariant: max 2 levels.
    pub fn check_depth(&self, parent: AgentId) -> Result<(), DepthError> {
        let depth = self.agent_depth(parent);
        if depth >= 2 {
            Err(DepthError::MaxDepthExceeded { current: depth })
        } else {
            Ok(())
        }
    }

    /// Enforce budget cascading: child ≤ remaining parent budget.
    pub fn allocate_child_budget(
        &self,
        parent: AgentId,
        requested: &DelegationBudget,
    ) -> Result<DelegationBudget, BudgetError> {
        let parent_remaining = self.remaining_budget(parent)?;
        if !parent_remaining.can_cover(requested) {
            return Err(BudgetError::InsufficientParentBudget);
        }
        Ok(requested.clone().cap_at(&parent_remaining))
    }
}

/// Full approval lease (P3 extension of P1 basic lease).
pub struct ApprovalLease {
    pub id:               Uuid,
    pub capability:       CapabilityName,
    pub scope:            LeaseScope,
    pub constraints:      LeaseConstraints,
    pub valid_until:      DateTime<Utc>,
    pub max_executions:   Option<u32>,
    pub executions_used:  u32,
    pub granted_by:       GrantSource,
    pub granted_at:       DateTime<Utc>,
    pub last_used_at:     Option<DateTime<Utc>>,
    pub revoked:          bool,
    pub audit_trail:      Vec<LeaseUseRecord>,
}

pub struct LeaseScope {
    pub account:          Option<String>,
    pub context_filters:  HashMap<String, serde_json::Value>,
}

pub struct LeaseConstraints {
    pub forbid: Vec<String>,     // e.g. ["delete"]
    pub allow:  Option<Vec<String>>,  // if Some, allowlist mode
}

pub enum GrantSource {
    UserExplicit,
    BiometricConfirm,
    SystemDefault,
}

pub struct LeaseUseRecord {
    pub intent_id:    Uuid,
    pub used_at:      DateTime<Utc>,
    pub arguments:    serde_json::Value,
    pub result:       ActionStatus,
}

impl ApprovalLease {
    /// Evaluate whether this lease authorizes a given intent.
    pub fn evaluate(&self, intent: &ActionIntent, now: DateTime<Utc>) -> LeaseDecision {
        if self.revoked {
            return LeaseDecision::Denied(LeaseReason::Revoked);
        }
        if now >= self.valid_until {
            return LeaseDecision::Denied(LeaseReason::Expired);
        }
        if let Some(max) = self.max_executions {
            if self.executions_used >= max {
                return LeaseDecision::Denied(LeaseReason::BudgetExhausted);
            }
        }
        if !self.scope.matches(intent) {
            return LeaseDecision::Denied(LeaseReason::OutOfScope);
        }
        if !self.constraints.permits(intent) {
            return LeaseDecision::Denied(LeaseReason::ConstraintViolation);
        }
        LeaseDecision::Authorized
    }

    /// Record a use. Increments counter, appends audit entry.
    pub fn record_use(&mut self, intent_id: Uuid, args: &serde_json::Value, result: ActionStatus) {
        self.executions_used += 1;
        self.last_used_at = Some(Utc::now());
        self.audit_trail.push(LeaseUseRecord {
            intent_id,
            used_at: Utc::now(),
            arguments: args.clone(),
            result,
        });
    }

    /// Revoke immediately. In-flight actions complete; no new ones.
    pub fn revoke(&mut self) {
        self.revoked = true;
    }
}

pub enum LeaseDecision {
    Authorized,
    Denied(LeaseReason),
}

pub enum LeaseReason {
    Revoked,
    Expired,
    BudgetExhausted,
    OutOfScope,
    ConstraintViolation,
}

9. ActionPipeline — Full Lifecycle Rust · waywiser-security

Blueprint §§17–20. The complete intent-to-receipt pipeline, bringing every P3 security component together.

Pipeline state machine

/// Lifecycle states of an ActionIntent through the security pipeline.
///
///   Proposed → Authorized → Executing → Verifying → Completed
///                 │              │            │
///                 ├→ Denied      ├→ Failed    ├→ UnknownSideEffect
///                 │              └→ Crashed   └→ Unexpected
///                 └→ Escalated (ask user)
#[derive(Copy, Clone, Debug)]
pub enum PipelineStage {
    Proposed,
    CapabilityCheck,
    RiskClassification,
    PolicyCheck,
    LeaseCheck,
    OsPermissionCheck,
    UserApproval,
    Authorized,
    Executing,
    Verifying,
    Completed,
    Denied { reason: String },
    Failed { error: String },
    UnknownSideEffect,
}

/// The full pipeline executor.
pub struct ActionPipeline {
    capabilities:   CapabilityRegistry,     // from P1
    classifier:     RiskClassifier,         // this phase
    policy:         WaywiserPolicy,         // from P1
    leases:         LeaseStore,             // this phase (full)
    os_permissions: OsPermissionChecker,    // from P1
}

impl ActionPipeline {
    /// Process an ActionIntent through the full security pipeline.
    /// Returns an ActionReceipt (always — even for denials).
    pub async fn process(
        &self,
        intent: ActionIntent,
        bridge: Option<&dyn A11yBridge>,
        user_prompt: &dyn UserApprovalPrompt,
    ) -> ActionReceipt {
        let mut stage = PipelineStage::Proposed;
        let started_at = Utc::now();

        // 1. Capability exists?
        stage = PipelineStage::CapabilityCheck;
        let spec = match self.capabilities.get(&intent.capability) {
            Some(s) => s,
            None => return ActionReceipt::denied(
                &intent, started_at, "unknown capability"
            ),
        };

        // 2. Risk classification (5-layer for UI actions, spec-declared for others)
        stage = PipelineStage::RiskClassification;
        let risk = if let Some(bridge) = bridge {
            // UI action: run 5-layer classifier
            let req = ClassificationRequest::from_intent(&intent, bridge);
            self.classifier.classify(&req).final_risk
        } else {
            // Non-UI: use capability-declared risk
            spec.risk
        };

        // 3. Waywiser policy check
        stage = PipelineStage::PolicyCheck;
        if let Err(deny) = self.policy.check(&intent, &spec, risk) {
            return ActionReceipt::denied(&intent, started_at, &deny.reason);
        }

        // 4. Approval lease check
        stage = PipelineStage::LeaseCheck;
        let lease_ok = self.leases.find_valid_lease(
            &intent.capability, &intent
        );

        // 5. OS permission check
        stage = PipelineStage::OsPermissionCheck;
        if let Err(e) = self.os_permissions.check_for(&spec) {
            return ActionReceipt::denied(&intent, started_at,
                &format!("OS permission missing: {e}"));
        }

        // 6. User approval if required and no valid lease
        stage = PipelineStage::UserApproval;
        if lease_ok.is_none() && risk.requires_user_approval() {
            match user_prompt.request_approval(&intent, risk).await {
                Approval::Granted => {}
                Approval::GrantedWithLease(lease) => {
                    self.leases.store(lease);
                }
                Approval::Denied => {
                    return ActionReceipt::denied(
                        &intent, started_at, "user denied"
                    );
                }
            }
        }

        // Record lease use if applicable
        if let Some(lease) = lease_ok {
            self.leases.record_use(lease.id, &intent);
        }

        // 7. Execute — this is the side-effect boundary
        stage = PipelineStage::Executing;

        // PERSIST intent as EXECUTING before dispatch (crash safety)
        self.persist_stage(&intent, PipelineStage::Executing);

        let exec_result = match spec.execution_mode {
            ExecutionMode::Local => self.execute_local(&intent, &spec).await,
            ExecutionMode::UiAutomation => {
                self.execute_ui(&intent, &spec, bridge.unwrap()).await
            }
            ExecutionMode::Remote => self.execute_remote(&intent, &spec).await,
        };

        // 8. Verification
        stage = PipelineStage::Verifying;
        let verification = match (&exec_result, bridge) {
            (Ok(_), Some(b)) => verify_after_action(b, &intent.expected_transition()),
            (Ok(_), None)    => VerificationResult::Unknown,
            (Err(_), _)      => VerificationResult::Unknown,
        };

        // 9. Build receipt
        let status = match exec_result {
            Ok(result) => ActionStatus::Completed,
            Err(ExecError::Crashed) => ActionStatus::UnknownSideEffect,
            Err(ExecError::Failed(e)) => ActionStatus::Failed(e),
        };

        ActionReceipt {
            intent_id:          intent.id,
            capability:         intent.capability.clone(),
            started_at,
            completed_at:       Some(Utc::now()),
            status,
            external_reference: exec_result.ok().and_then(|r| r.external_ref),
            reversible:         spec.reversible,
            undo_token:         exec_result.ok().and_then(|r| r.undo_token),
            verification,
            result_summary:     exec_result.ok().map(|r| r.summary),
        }
    }
}

/// Replay policy enforcement on crash recovery.
impl ActionPipeline {
    pub fn recover_after_crash(
        &self,
        pending: Vec<ActionIntent>,
    ) -> Vec<RecoveryAction> {
        pending.into_iter().map(|intent| {
            let spec = self.capabilities.get(&intent.capability);
            match spec.map(|s| s.replay_policy) {
                Some(ReplayPolicy::SafeReplay) =>
                    RecoveryAction::Retry(intent),
                Some(ReplayPolicy::NeverReplay) =>
                    RecoveryAction::MarkFailed(intent, "never-replay policy"),
                Some(ReplayPolicy::VerifyBeforeRetry) =>
                    RecoveryAction::VerifyThenDecide(intent),
                None =>
                    RecoveryAction::MarkFailed(intent, "unknown capability"),
            }
        }).collect()
    }
}

pub enum RecoveryAction {
    Retry(ActionIntent),
    MarkFailed(ActionIntent, &'static str),
    VerifyThenDecide(ActionIntent),
}

10. RemoteExecution Rust · waywiser-remote — OPTIONAL

Optional remote execution nodes for delegating heavy workloads. Ships after the rest of P3. Remote nodes are NOT trusted more than local — same security kernel applies.

/// A remote compute node that can execute capabilities.
pub struct RemoteNode {
    pub id:             Uuid,
    pub name:           String,
    pub endpoint:       String,
    pub capabilities:   Vec<CapabilityName>,
    pub status:         RemoteNodeStatus,
    pub last_heartbeat: DateTime<Utc>,
    pub auth_token:     String,
}

#[derive(Copy, Clone, Debug)]
pub enum RemoteNodeStatus {
    Online,
    Degraded,
    Offline,
}

/// Remote capability — proxied through the local security kernel.
/// Security is NOT delegated to the remote node.
pub struct RemoteExecutor {
    nodes:  Vec<RemoteNode>,
    kernel: Arc<SecurityKernel>,
}

impl RemoteExecutor {
    /// Execute on remote. Full local authorization first.
    pub async fn execute(
        &self,
        intent: &ActionIntent,
    ) -> Result<ActionReceipt, RemoteError> {
        // Local authorization FIRST (same pipeline as local execution)
        // Only the execution step is delegated to remote.
        let node = self.find_capable_node(&intent.capability)?;
        if node.status == RemoteNodeStatus::Offline {
            return Err(RemoteError::NodeOffline(node.id));
        }

        let result = self.dispatch_to_node(&node, intent).await;

        match result {
            Ok(receipt) => {
                // Verify result locally
                Ok(receipt)
            }
            Err(RemoteError::Disconnected) => {
                // In-flight → UNKNOWN_SIDE_EFFECT
                Ok(ActionReceipt {
                    status: ActionStatus::UnknownSideEffect,
                    verification: VerificationResult::Unknown,
                    ..ActionReceipt::from_intent(intent)
                })
            }
            Err(e) => Err(e),
        }
    }
}

Acceptance Criteria

  1. ECM provisioned: AccessibilityService enabled via MDM on target device → tree inspection works, nodes readable for all non-secure apps.
  2. Banking blocked: Automation attempted on com.example.bank → RiskClassifier returns Block, no action taken, no override possible regardless of lease or user request.
  3. Reviewed profile enforced: Company mail send button → reviewed profile maps to Communication risk → user approval required → approved → action executes → receipt logged.
  4. Unknown app fallback: Click on unknown app → classified as DeviceControl → user prompted → audit trail records classification layer trace.
  5. Risk monotonicity: 500 synthetic UI actions through classifier → risk NEVER decreases between layers. Automated property test.
  6. TOCTOU protection: Target button moves between plan and execution → fingerprint mismatch → abort with TocTouError::FingerprintMismatch, no side effect executed.
  7. High-risk TOCTOU strictness: Financial-risk action with FingerprintMatch::Partial → rejected (exact match required for high risk).
  8. Post-action halt on Unexpected: Click send → unexpected error dialog appears → VerificationResult::Unexpected → all further workflow steps halted immediately.
  9. Cross-app workflow: "Reschedule dentist" → open calendar → find event → modify → verify → receipt with undo token. Receipts match actual calendar state.
  10. Workflow rollback: Workflow step 3 of 5 fails → steps 1–2 (reversible) rolled back via undo tokens → rolled-back steps marked RolledBack.
  11. Lease lifecycle: Granted for calendar.update until 18:00, max 5 → works 5 times → 6th denied with BudgetExhausted → re-approval required.
  12. Lease expiry: Lease expires → immediate denial, clean LeaseReason::Expired. No grace period.
  13. Self-improvement blocked: Learned skill attempts waywiser.security.grant_capability → deterministic validator rejects, invariant I10 enforced.
  14. Delegation depth: Orchestrator creates 2 leaf agents → succeeds. Leaf attempts to delegate → denied with DepthError::MaxDepthExceeded.
  15. Budget cascading: Parent has 1000 tokens remaining → child requests 1500 → capped at 1000.
  16. Verification agent read-only: Verification agent attempts a write capability → denied by CapabilityFilter::ReadOnly.
  17. Crash recovery: Process crash after calendar update dispatch → recovery finds PipelineStage::Executing → VerifyBeforeRetry policy → verifies calendar state → correct state determined, receipt updated.
  18. FLAG_SECURE: Secure window detected → SecureWindowState::SecureBlocked → no fabricated visual context, automation gracefully declined.
  19. Idempotency: Same idempotency_key submitted twice → second execution skipped, original receipt returned.
  20. Soak test: 100 cross-app actions over 8 hours → all receipts match actual app outcomes. No leaked JNI refs, no memory growth, no orphaned intents.
