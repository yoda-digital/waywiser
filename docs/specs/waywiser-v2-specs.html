AgentLoop

/// Core agent cycle: prepare context → send to model → process response.
#[async_trait]
pub trait AgentLoop: Send + Sync {
    /// Build the context window for the next model call.
    async fn prepare(&self, session: &SessionState) -> Result<PreparedContext, WaywiserError>;

    /// Send prepared context to the inference backend, streaming tokens back.
    async fn execute(
        &self,
        ctx: PreparedContext,
        backend: &dyn InferenceBackend,
        cancel: CancellationToken,
    ) -> Result<ExecutionResult, WaywiserError>;

    /// Apply the model's response: persist records, enqueue follow-ups, run tools.
    async fn finalize(
        &self,
        result: ExecutionResult,
        session: &mut SessionState,
        store: &dyn SessionBackend,
    ) -> Result<FinalizeOutcome, WaywiserError>;
}

#[derive(Debug)]
pub enum FinalizeOutcome {
    /// Turn complete, no pending work.
    Done,
    /// Model requested tool calls; loop must execute them and re-enter.
    ToolCallsPending(Vec<ToolCall>),
    /// Model requested a follow-up turn.
    FollowUp,
    /// Steering request arrived mid-turn; re-prepare.
    Steered(SteerRequest),
    /// Abort requested.
    Aborted,
}

Key Behaviors

  • The loop drains the lane queue after every FinalizeOutcome::Done: pending_steer > pending_follow_up > next_run > deferred (priority order).
  • Steer requests arriving during Executing are queued, not applied immediately.
  • Recovery attempts up to 3 times with escalating repair: (1) re-read from store, (2) rebuild lane from records, (3) truncate corrupted tail.
  • All state transitions are logged as OperationRecord entries before the transition occurs (write-ahead).

Error Types

Uses WaywiserError::SessionNotFound, SessionCorrupted, CompactionFailed.

Testing

Conformance suite (component 7) drives all agent loop tests. Additionally: property tests for reducer state machine (no invalid transition reachable), fuzz OperationRecord deserialization.

Implementation Notes

  • WAL journal mode for concurrent reads during writes.
  • Single writer, multiple readers via r2d2 connection pool.
  • Schema migrations via embedded SQL files with version tracking table.
  • Session state serialized as CBOR blobs for lanes/entries; records stored row-per-record for query.
  • All writes within a single SQLite transaction per save_session call.

// Repository bridging Rust events to Compose StateFlow.
class WaywiserRepository(private val runtime: WaywiserRuntime) {

    private val _events = MutableSharedFlow<RuntimeEvent>(
        extraBufferCapacity = 64,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val events: SharedFlow<RuntimeEvent> = _events

    private var pollJob: Job? = null

    fun startPolling(scope: CoroutineScope) {
        pollJob = scope.launch(Dispatchers.IO) {
            while (isActive) {
                try {
                    val event = runtime.pollEvent() ?: continue
                    _events.emit(event)
                } catch (e: WaywiserException) {
                    _events.emit(RuntimeEvent.Error(e.code, e.message ?: "unknown"))
                }
            }
        }
    }

    fun stopPolling() { pollJob?.cancel() }

    suspend fun sendMessage(content: String) = withContext(Dispatchers.IO) {
        runtime.sendMessage(content)
    }

    suspend fun cancel() = withContext(Dispatchers.IO) {
        runtime.cancel()
    }
}

Mandatory FFI Discipline

  • std::panic::catch_unwind wraps EVERY UniFFI-exported function. CI lint (clippy custom lint or build script) fails the build if any exported fn lacks it.
  • Cargo.toml: panic = "unwind" (never "abort").
  • Megazord pattern: single libwaywiser.so containing all Rust crates.
  • Kotlin callbacks (if any) use weak references; uniffiIsDestroyed checked before invoking.

Authorization Flow

/// The authorization pipeline — each step can only maintain or INCREASE risk
impl SecurityKernel {
    pub fn authorize(&self, intent: &ActionIntent) -> SecurityDecision {
        // Step 1: Capability exists?
        let cap = match self.registry.get(&intent.capability) {
            Some(c) => c,
            None => return SecurityDecision::Denied(
                DenialReason::UnknownCapability(intent.capability.clone())
            ),
        };

        // Step 2: Risk classification (from CapabilitySpec)
        let risk = cap.risk;

        // Step 3: Waywiser policy check
        if let Some(denial) = self.policy.check(intent, &cap) {
            return SecurityDecision::Denied(denial);
        }

        // Step 4: OS permission available?
        if let Some(os_perm) = &cap.os_permission {
            if !self.os_permissions.is_granted(os_perm) {
                return SecurityDecision::Denied(
                    DenialReason::OsPermissionMissing(os_perm.clone())
                );
            }
        }

        // Step 5: Active approval lease?
        if let Some(lease) = self.find_valid_lease(intent) {
            if lease.executions_used < lease.max_executions {
                return SecurityDecision::Allowed(AuthorizationSource::Lease(lease.id));
            }
            // Lease exhausted — fall through to approval required
        }

        // Step 6: Determine approval kind based on risk
        match risk {
            RiskLevel::ReadPersonal => SecurityDecision::Allowed(
                AuthorizationSource::DefaultPolicy
            ),
            RiskLevel::DeviceControl
            | RiskLevel::CrossAppWrite => SecurityDecision::RequiresApproval(
                ApprovalKind::UserConfirm
            ),
            RiskLevel::Communication
            | RiskLevel::Financial
            | RiskLevel::Destructive => SecurityDecision::RequiresApproval(
                ApprovalKind::BiometricConfirm
            ),
            RiskLevel::UiUnclassifiedWrite => SecurityDecision::RequiresApproval(
                ApprovalKind::UserConfirm
            ),
        }
    }
}

impl AttentionGovernor {
    /// Evaluate attention for a signal.
    /// Pipeline: floor → ceiling → deterministic match → edge hint → final.
    pub fn evaluate(
        &self,
        signal: &AttentionSignal,
        edge_hint: Option<&EdgeHint>,
        context: &ContextGraphSnapshot,
    ) -> AttentionResult {
        // 1. Find matching floor (strongest wins)
        let floor = self.resolve_floor(signal);

        // 2. Find matching ceiling
        let ceiling = self.resolve_ceiling(signal);

        // 3. Deterministic pattern match → base decision
        let base = self.deterministic_classify(signal, context);

        // 4. Apply edge hint (ONLY escalation; suppression requires
        //    confidence > suppression_threshold AND no floor violation)
        let adjusted = match edge_hint {
            Some(hint) => self.apply_hint(base, hint, floor),
            None => base,
        };

        // 5. Clamp to [floor, ceiling]
        let final_decision = self.clamp(adjusted, floor, ceiling);

        AttentionResult {
            decision: final_decision,
            floor_applied: floor,
            ceiling_applied: ceiling,
            edge_hint_effect: edge_hint.map(|h| h.suggested),
            source_rule: self.matched_rule(signal),
        }
    }

    /// Suppression asymmetry: escalation easier than suppression.
    fn apply_hint(
        &self,
        base: AttentionDecision,
        hint: &EdgeHint,
        floor: Option<AttentionDecision>,
    ) -> AttentionDecision {
        if hint.suggested > base {
            // Escalation: lower confidence threshold
            if hint.confidence >= self.policy.escalation_threshold {
                hint.suggested
            } else {
                base
            }
        } else if hint.suggested < base {
            // Suppression: higher confidence threshold + floor check
            if hint.confidence >= self.policy.suppression_threshold {
                // NEVER suppress below floor
                match floor {
                    Some(f) if hint.suggested < f => f,
                    _ => hint.suggested,
                }
            } else {
                base // insufficient confidence to suppress
            }
        } else {
            base
        }
    }
}

pub struct AttentionResult {
    pub decision: AttentionDecision,
    pub floor_applied: Option<AttentionDecision>,
    pub ceiling_applied: Option<AttentionDecision>,
    pub edge_hint_effect: Option<AttentionDecision>,
    pub source_rule: Option<AttentionRule>,
}

impl ProjectionEngine {
    /// Project relevant context for the current query.
    pub fn project(
        &self,
        snapshot: &ContextGraphSnapshot,
        query: &str,
        task_type: TaskType,
        budget_tokens: usize,
    ) -> ContextProjection {
        // 1. Score each node for relevance to query + task
        let scored: Vec<(f32, &ContextNode)> = snapshot.nodes.iter()
            .map(|n| (self.relevance_score(n, query, task_type), n))
            .collect();

        // 2. Sort by score descending
        // 3. Greedily include nodes until budget exhausted
        // 4. Summarize verbose nodes to fit budget

        let mut projection = ContextProjection::new();
        let mut tokens_used = 0;
        for (score, node) in scored {
            let node_tokens = self.token_estimator.estimate(node);
            if tokens_used + node_tokens > budget_tokens {
                // Try summarizing
                if let Some(summary) = self.summarize(node) {
                    let summary_tokens = self.token_estimator.estimate_str(&summary);
                    if tokens_used + summary_tokens <= budget_tokens {
                        projection.add_summary(summary, score);
                        tokens_used += summary_tokens;
                    }
                }
                continue;
            }
            projection.add_node(node.clone(), score);
            tokens_used += node_tokens;
        }

        projection.tokens_used = tokens_used;
        projection
    }
}

impl ReasoningQueueManager {
    /// Enqueue a new job, coalescing with existing if key matches.
    pub async fn enqueue(&self, job: ReasoningJob) -> Result<Uuid, QueueError> {
        if let Some(key) = &job.coalesce_key {
            if let Some(existing) = self.store.find_by_coalesce_key(key).await? {
                // Merge: keep earlier job, update context_refs
                self.store.merge_context_refs(existing.id, &job.context_refs).await?;
                return Ok(existing.id);
            }
        }
        self.store.insert(&job).await
    }

    /// Pick next job to execute. Checks relevance before returning.
    pub async fn next_ready(&self) -> Result<Option<ReasoningJob>, QueueError> {
        loop {
            let candidate = self.store.next_by_priority().await?;
            match candidate {
                None => return Ok(None),
                Some(job) => {
                    if self.is_still_relevant(&job).await {
                        return Ok(Some(job));
                    } else {
                        self.store.mark_expired(
                            job.id,
                            "Cause no longer relevant in context graph",
                        ).await?;
                    }
                }
            }
        }
    }

    /// Semantic relevance check — not just "older than N minutes".
    async fn is_still_relevant(&self, job: &ReasoningJob) -> bool {
        match &job.relevance_rule {
            RelevanceRule::Always => true,
            RelevanceRule::BeforeDeadline(dl) => Utc::now() < *dl,
            RelevanceRule::ContextNodePresent { domain_key, .. } => {
                let graph = self.context_graph.read().await;
                graph.has_node(domain_key)
            }
            RelevanceRule::Custom(_) => true, // custom eval deferred
        }
    }
}

    let with_desc = count_with(root, |n| {
        n.content_description.is_some() || n.text.is_some()

    match ratio {
        r if r >= 0.6 => TreeQuality::Good,
        r if r >= 0.3 => TreeQuality::Partial,
        r if r >= 0.1 => TreeQuality::Poor,
        _              => TreeQuality::Unusable,
    }

    Paste,
    Gesture,

impl RiskClassifier {
    /// Classify risk. Layers execute in order 1→5.
    /// Risk floor can only increase; decrease is a logic error.
    pub fn classify(&self, req: &ClassificationRequest) -> ClassificationResult {
        let mut risk = RiskLevel::None;
        let mut trace = Vec::new();

        // Layer 1: Sensitive-package policy
        if let Some(policy) = self.sensitive_packages.get(&req.package) {
            if policy.automation == AutomationPolicy::Block {
                return ClassificationResult::blocked(
                    &trace, "sensitive-package: automation blocked"
                );
            }
            if let Some(floor) = policy.floor {
                risk = risk.max(floor);
            }
            trace.push(LayerDecision::hit(ClassificationLayer::SensitivePackage, risk));
        }

        // Layer 2: Reviewed automation profile
        if let Some(profile) = self.find_profile(&req.package, &req.node) {
            risk = risk.max(profile.risk);
            trace.push(LayerDecision::hit(ClassificationLayer::ReviewedProfile, risk));
        }

        // Layer 3: Deterministic semantic rules
        let text = req.node.combined_text();
        for rule in &self.semantic_rules {
            if rule.pattern.is_match(&text) {
                risk = risk.max(rule.floor);
                trace.push(LayerDecision::hit(ClassificationLayer::SemanticRule, risk));
                break;
            }
        }

        // Layer 4: Primitive action-type floor
        if let Some(&floor) = self.primitive_floors.get(&req.action) {
            risk = risk.max(floor);
            trace.push(LayerDecision::hit(ClassificationLayer::PrimitiveActionFloor, risk));
        }

        // Layer 5: Unknown fallback
        if risk == RiskLevel::None {
            risk = RiskLevel::UiUnclassifiedWrite;
            trace.push(LayerDecision::hit(ClassificationLayer::UnknownFallback, risk));
        }

        // Log LLM hint but NEVER let it lower risk
        if let Some(hint) = &req.llm_hint {
            tracing::info!(llm_hint = ?hint, final_risk = ?risk,
                "LLM risk hint logged (non-authoritative)");
        }

        ClassificationResult {
            final_risk: risk,
            layer_trace: trace,
            llm_hint_used: false,
            approval_mode: risk.default_approval_mode(),
        }
    }
}

/// Policy: what FingerprintMatch is acceptable per risk level?
pub fn acceptable_match(risk: RiskLevel, m: &FingerprintMatch) -> bool {
    match (risk, m) {
        (_, FingerprintMatch::Exact) => true,
        (_, FingerprintMatch::NoMatch) => false,
        // High risk: require Exact — no partial allowed
        (RiskLevel::Financial | RiskLevel::Destructive | RiskLevel::Communication,
         FingerprintMatch::Partial { .. }) => false,
        // Lower risk: partial OK if mismatches are cosmetic
        (_, FingerprintMatch::Partial { mismatches }) =>
            mismatches.iter().all(|m| m.is_cosmetic()),
    }
}

/// Pre-action verification: re-resolve, fingerprint, decide.
pub fn verify_before_action(
    bridge: &dyn A11yBridge,
    planned: &NodeFingerprint,
    risk: RiskLevel,
) -> Result<A11yNode, TocTouError> {
    let snapshot = bridge.capture_tree_snapshot(planned.window_id)?;
    let resolved = find_by_fingerprint(&snapshot.root, planned)
        .ok_or(TocTouError::TargetNotFound)?;
    let current_fp = NodeFingerprint::from_node(&resolved);
    let comparison = planned.compare(&current_fp);
    if acceptable_match(risk, &comparison) {
        Ok(resolved)
    } else {
        Err(TocTouError::FingerprintMismatch {
            planned: planned.clone(),
            actual: current_fp,
            comparison,
        })
    }
}

    FingerprintMismatch {
        planned:    NodeFingerprint,

    SecureWindow,

/// Post-action verification. Called after every significant UI action.
pub fn verify_after_action(
    bridge: &dyn A11yBridge,
    expected: &ExpectedTransition,
) -> VerificationResult {
    // Wait briefly for transition to settle
    std::thread::sleep(expected.verification_timeout.min(Duration::from_millis(500)));

    let snapshot = match bridge.capture_tree_snapshot_best_effort() {
        Ok(s)  => s,
        Err(_) => return VerificationResult::Unknown,
    };

    let indicators_present = expected.expected_indicators.iter()
        .filter(|ind| find_indicator(&snapshot.root, ind).is_some())
        .count();

    let removals_gone = expected.expected_removals.iter()
        .filter(|rid| find_by_resource_id(&snapshot.root, rid).is_none())
        .count();

    let total_checks = expected.expected_indicators.len()
        + expected.expected_removals.len();

    if total_checks == 0 {
        return VerificationResult::Unknown;
    }

    let passed = indicators_present + removals_gone;
    let ratio = passed as f64 / total_checks as f64;

    match ratio {
        r if r >= 0.8 => VerificationResult::Verified,
        r if r >= 0.4 => VerificationResult::Likely,
        _              => VerificationResult::Unexpected,
    }
}

impl Workflow {
    /// Execute the workflow step-by-step.
    /// Between each step: verify pre-conditions and post-action outcome.
    /// On UNEXPECTED: halt and record partial completion.
    pub async fn execute(
        &mut self,
        kernel: &SecurityKernel,
        bridge: &dyn A11yBridge,
    ) -> WorkflowResult {
        for step in &mut self.steps {
            // 1. Check pre-conditions
            if !step.check_pre_conditions() {
                step.status = StepStatus::Failed;
                self.status = WorkflowStatus::PartiallyCompleted {
                    failed_at: step.index
                };
                return WorkflowResult::Partial;
            }

            // 2. Run through full ActionPipeline
            let receipt = kernel.process_intent(&step.action_intent).await;

            // 3. Post-action verification
            let verification = verify_after_action(bridge, &step.expected_outcome);
            step.actual_outcome = Some(verification);
            step.receipt = Some(receipt);

            match verification {
                VerificationResult::Unexpected => {
                    step.status = StepStatus::Failed;
                    self.status = WorkflowStatus::Aborted {
                        reason: AbortReason::VerificationUnexpected
                    };
                    // HALT — do not execute further steps
                    return WorkflowResult::Aborted;
                }
                VerificationResult::Unknown => {
                    step.status = StepStatus::UnknownSideEffect;
                    // Continue with caution — next step checks pre-conditions
                }
                _ => {
                    step.status = StepStatus::Succeeded;
                }
            }
        }
        self.status = WorkflowStatus::Completed;
        WorkflowResult::Completed
    }

    /// Attempt to roll back completed reversible steps.
    pub async fn rollback(
        &mut self,
        kernel: &SecurityKernel,
    ) -> RollbackResult {
        let mut rolled = 0;
        for step in self.steps.iter_mut().rev() {
            if step.status != StepStatus::Succeeded { continue; }
            if let Some(receipt) = &step.receipt {
                if receipt.reversible && receipt.undo_token.is_some() {
                    // Execute compensating action
                    kernel.execute_undo(receipt).await;
                    step.status = StepStatus::RolledBack;
                    rolled += 1;
                }
            }
        }
        RollbackResult { steps_rolled_back: rolled }
    }
}

            Some(s) => s,
            None => return ActionReceipt::denied(
                &intent, started_at, "unknown capability"
            ),
        };

        // 2. Risk classification (5-layer for UI actions, spec-declared for others)
        stage = PipelineStage::RiskClassification;
        let risk = if let Some(bridge) = bridge {
            // UI action: run 5-layer classifier
            let req = ClassificationRequest::from_intent(&intent, bridge);
            self.classifier.classify(&req).final_risk
        } else {
            // Non-UI: use capability-declared risk
            spec.risk
        };

        // 3. Waywiser policy check
        stage = PipelineStage::PolicyCheck;
        if let Err(deny) = self.policy.check(&intent, &spec, risk) {
            return ActionReceipt::denied(&intent, started_at, &deny.reason);
        }

        // 4. Approval lease check
        stage = PipelineStage::LeaseCheck;
        let lease_ok = self.leases.find_valid_lease(
            &intent.capability, &intent
        );

        // 5. OS permission check
        stage = PipelineStage::OsPermissionCheck;
        if let Err(e) = self.os_permissions.check_for(&spec) {
            return ActionReceipt::denied(&intent, started_at,
                &format!("OS permission missing: {e}"));
        }

        // 6. User approval if required and no valid lease
        stage = PipelineStage::UserApproval;
        if lease_ok.is_none() && risk.requires_user_approval() {
            match user_prompt.request_approval(&intent, risk).await {
                Approval::Granted => {}
                Approval::GrantedWithLease(lease) => {
                    self.leases.store(lease);
                }
                Approval::Denied => {
                    return ActionReceipt::denied(
                        &intent, started_at, "user denied"
                    );
                }
            }
        }

        // Record lease use if applicable
        if let Some(lease) = lease_ok {
            self.leases.record_use(lease.id, &intent);
        }

        // 7. Execute — this is the side-effect boundary
        stage = PipelineStage::Executing;

        // PERSIST intent as EXECUTING before dispatch (crash safety)
        self.persist_stage(&intent, PipelineStage::Executing);

        let exec_result = match spec.execution_mode {
            ExecutionMode::Local => self.execute_local(&intent, &spec).await,
            ExecutionMode::UiAutomation => {
                self.execute_ui(&intent, &spec, bridge.unwrap()).await
            }
            ExecutionMode::Remote => self.execute_remote(&intent, &spec).await,
        };

        // 8. Verification
        stage = PipelineStage::Verifying;
        let verification = match (&exec_result, bridge) {
            (Ok(_), Some(b)) => verify_after_action(b, &intent.expected_transition()),
            (Ok(_), None)    => VerificationResult::Unknown,
            (Err(_), _)      => VerificationResult::Unknown,
        };

        // 9. Build receipt
        let status = match exec_result {
            Ok(result) => ActionStatus::Completed,
            Err(ExecError::Crashed) => ActionStatus::UnknownSideEffect,
            Err(ExecError::Failed(e)) => ActionStatus::Failed(e),
        };

        ActionReceipt {
            intent_id:          intent.id,
            capability:         intent.capability.clone(),
            started_at,
            completed_at:       Some(Utc::now()),
            status,
            external_reference: exec_result.ok().and_then(|r| r.external_ref),
            reversible:         spec.reversible,
            undo_token:         exec_result.ok().and_then(|r| r.undo_token),
            verification,
            result_summary:     exec_result.ok().map(|r| r.summary),
        }
    }
}

/// Replay policy enforcement on crash recovery.
impl ActionPipeline {
    pub fn recover_after_crash(
        &self,
        pending: Vec<ActionIntent>,
    ) -> Vec<RecoveryAction> {
        pending.into_iter().map(|intent| {
            let spec = self.capabilities.get(&intent.capability);
            match spec.map(|s| s.replay_policy) {
                Some(ReplayPolicy::SafeReplay) =>
                    RecoveryAction::Retry(intent),
                Some(ReplayPolicy::NeverReplay) =>
                    RecoveryAction::MarkFailed(intent, "never-replay policy"),
                Some(ReplayPolicy::VerifyBeforeRetry) =>
                    RecoveryAction::VerifyThenDecide(intent),
                None =>
                    RecoveryAction::MarkFailed(intent, "unknown capability"),
            }
        }).collect()
    }
}

pub enum RecoveryAction {
    Retry(ActionIntent),
    MarkFailed(ActionIntent, &'static str),
    VerifyThenDecide(ActionIntent),
}

/// Security is NOT delegated to the remote node.

    nodes:  Vec<RemoteNode>,
    kernel: Arc<SecurityKernel>,

impl RemoteExecutor {
    /// Execute on remote. Full local authorization first.
    pub async fn execute(
        &self,
        intent: &ActionIntent,
    ) -> Result<ActionReceipt, RemoteError> {
        // Local authorization FIRST (same pipeline as local execution)
        // Only the execution step is delegated to remote.
        let node = self.find_capable_node(&intent.capability)?;
        if node.status == RemoteNodeStatus::Offline {
            return Err(RemoteError::NodeOffline(node.id));
        }

        let result = self.dispatch_to_node(&node, intent).await;

        match result {
            Ok(receipt) => {
                // Verify result locally
                Ok(receipt)
            }
            Err(RemoteError::Disconnected) => {
                // In-flight → UNKNOWN_SIDE_EFFECT
                Ok(ActionReceipt {
                    status: ActionStatus::UnknownSideEffect,
                    verification: VerificationResult::Unknown,
                    ..ActionReceipt::from_intent(intent)
                })
            }
            Err(e) => Err(e),
        }
    }
}
