Observation bus & reducers

/// Receives observations from the Android layer via FFI.
pub struct ObservationBus {
    subscribers: Vec<Box<dyn DeterministicReducer>>,
}

impl ObservationBus {
    /// Dispatch an observation to all registered reducers.
    pub fn publish(&self, obs: &Observation, graph: &mut ContextGraph) {
        for reducer in &self.subscribers {
            reducer.reduce(obs, graph);
        }
    }
}

/// Deterministic reducer: pure function from observation to graph mutation.
/// No LLM calls. No network. No side effects.
pub trait DeterministicReducer: Send + Sync {
    /// Which observation kinds this reducer handles.
    fn handles(&self) -> &[ObservationKind];

    /// Apply the observation to the graph. Must be deterministic and fast.
    fn reduce(&self, obs: &Observation, graph: &mut ContextGraph);
}

pub struct AttentionGovernor {
    policy: AttentionPolicy,
}

impl AttentionGovernor {
    /// Evaluate attention for a signal.
    /// Pipeline: floor → ceiling → deterministic match → edge hint → final.
    pub fn evaluate(
        &self,
        signal: &AttentionSignal,
        edge_hint: Option<&EdgeHint>,
        context: &ContextGraphSnapshot,
    ) -> AttentionResult {
        // 1. Find matching floor (strongest wins)
        let floor = self.resolve_floor(signal);

        // 2. Find matching ceiling
        let ceiling = self.resolve_ceiling(signal);

        // 3. Deterministic pattern match → base decision
        let base = self.deterministic_classify(signal, context);

        // 4. Apply edge hint (ONLY escalation; suppression requires
        //    confidence > suppression_threshold AND no floor violation)
        let adjusted = match edge_hint {
            Some(hint) => self.apply_hint(base, hint, floor),
            None => base,
        };

        // 5. Clamp to [floor, ceiling]
        let final_decision = self.clamp(adjusted, floor, ceiling);

        AttentionResult {
            decision: final_decision,
            floor_applied: floor,
            ceiling_applied: ceiling,
            edge_hint_effect: edge_hint.map(|h| h.suggested),
            source_rule: self.matched_rule(signal),
        }
    }

    /// Suppression asymmetry: escalation easier than suppression.
    fn apply_hint(
        &self,
        base: AttentionDecision,
        hint: &EdgeHint,
        floor: Option<AttentionDecision>,
    ) -> AttentionDecision {
        if hint.suggested > base {
            // Escalation: lower confidence threshold
            if hint.confidence >= self.policy.escalation_threshold {
                hint.suggested
            } else {
                base
            }
        } else if hint.suggested < base {
            // Suppression: higher confidence threshold + floor check
            if hint.confidence >= self.policy.suppression_threshold {
                // NEVER suppress below floor
                match floor {
                    Some(f) if hint.suggested < f => f,
                    _ => hint.suggested,
                }
            } else {
                base // insufficient confidence to suppress
            }
        } else {
            base
        }
    }
}

pub struct AttentionResult {
    pub decision: AttentionDecision,
    pub floor_applied: Option<AttentionDecision>,
    pub ceiling_applied: Option<AttentionDecision>,
    pub edge_hint_effect: Option<AttentionDecision>,
    pub source_rule: Option<AttentionRule>,
}

/// Combines lexical (FTS5) and semantic (vector) recall.
pub struct HybridRecallEngine {
    lexical: LexicalIndex,       // from P0
    semantic: Box<dyn VectorIndex>,
    embedder: Box<dyn EmbeddingProvider>,
}

impl HybridRecallEngine {
    pub async fn recall(
        &self,
        query: &str,
        scope: &RecallScope,
        limit: usize,
    ) -> Result<Vec<RecallResult>, MemoryError> {
        // Run both in parallel
        let (lexical, semantic) = tokio::join!(
            self.lexical.search(query, scope, limit),
            self.semantic_search(query, scope, limit),
        );

        // Merge and rank by combined score
        // Weights: lexical_score * 0.3 + semantic_score * 0.3
        //        + recency * 0.15 + usage * 0.1 + confidence * 0.15
        self.merge_and_rank(lexical?, semantic?, limit)
    }

    /// Fail-soft: if embeddings unavailable, return lexical-only results.
    async fn semantic_search(
        &self, query: &str, scope: &RecallScope, limit: usize,
    ) -> Result<Vec<RecallResult>, MemoryError> {
        match self.embedder.embed(query).await {
            Ok(vec) => { /* vector search */ }
            Err(_) => Ok(vec![]),  // fail-soft: no semantic results
        }
    }
}

Vector migration protocol (blueprint §50)

1. Old index continues serving all queries
2. New index builds independently (background, P4 priority)
3. Validate: new index covers all entries, quality spot-check
4. Atomic switch: swap search target
5. Retire old index after grace period
  • Invariant: incompatible embedding spaces are NEVER mixed in a single search
  • Each VectorEntry carries embedding_model_id — entries from different models cannot co-exist in one index

8. ContextProjection waywiser-context (submodule)

The model never receives the complete context graph. Blueprint §13.

pub struct ProjectionEngine {
    token_estimator: Box<dyn TokenEstimator>,
}

impl ProjectionEngine {
    /// Project relevant context for the current query.
    pub fn project(
        &self,
        snapshot: &ContextGraphSnapshot,
        query: &str,
        task_type: TaskType,
        budget_tokens: usize,
    ) -> ContextProjection {
        // 1. Score each node for relevance to query + task
        let scored: Vec<(f32, &ContextNode)> = snapshot.nodes.iter()
            .map(|n| (self.relevance_score(n, query, task_type), n))
            .collect();

        // 2. Sort by score descending
        // 3. Greedily include nodes until budget exhausted
        // 4. Summarize verbose nodes to fit budget

        let mut projection = ContextProjection::new();
        let mut tokens_used = 0;
        for (score, node) in scored {
            let node_tokens = self.token_estimator.estimate(node);
            if tokens_used + node_tokens > budget_tokens {
                // Try summarizing
                if let Some(summary) = self.summarize(node) {
                    let summary_tokens = self.token_estimator.estimate_str(&summary);
                    if tokens_used + summary_tokens <= budget_tokens {
                        projection.add_summary(summary, score);
                        tokens_used += summary_tokens;
                    }
                }
                continue;
            }
            projection.add_node(node.clone(), score);
            tokens_used += node_tokens;
        }

        projection.tokens_used = tokens_used;
        projection
    }
}

    pub cause: ReasoningCause,
    pub created_at: DateTime<Utc>,
    pub priority: InferencePriority,
    pub context_refs: Vec<ContextRef>,

    // -- P2 extensions --
    pub relevance_rule: RelevanceRule,
    pub status: ReasoningJobStatus,
    pub attempts: u32,
    pub last_attempt_at: Option<DateTime<Utc>>,
    pub coalesce_key: Option<String>,  // jobs with same key merge
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum RelevanceRule {
    /// Still relevant if this context node exists and matches.
    ContextNodePresent { domain_key: ContextDomainKey, predicate: String },
    /// Still relevant if time hasn't passed a deadline.
    BeforeDeadline(DateTime<Utc>),
    /// Always relevant (execute when capacity available).
    Always,
    /// Custom: evaluate via a deterministic function.
    Custom(String),
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum ReasoningJobStatus {
    Pending,
    Executing,
    Completed(serde_json::Value),
    Expired { reason: String },
    Failed { error: String, retryable: bool },
}

Queue manager

pub struct ReasoningQueueManager {
    store: Box<dyn ReasoningJobStore>,  // SQLite-backed, survives process death
    context_graph: Arc<RwLock<ContextGraph>>,
}

impl ReasoningQueueManager {
    /// Enqueue a new job, coalescing with existing if key matches.
    pub async fn enqueue(&self, job: ReasoningJob) -> Result<Uuid, QueueError> {
        if let Some(key) = &job.coalesce_key {
            if let Some(existing) = self.store.find_by_coalesce_key(key).await? {
                // Merge: keep earlier job, update context_refs
                self.store.merge_context_refs(existing.id, &job.context_refs).await?;
                return Ok(existing.id);
            }
        }
        self.store.insert(&job).await
    }

    /// Pick next job to execute. Checks relevance before returning.
    pub async fn next_ready(&self) -> Result<Option<ReasoningJob>, QueueError> {
        loop {
            let candidate = self.store.next_by_priority().await?;
            match candidate {
                None => return Ok(None),
                Some(job) => {
                    if self.is_still_relevant(&job).await {
                        return Ok(Some(job));
                    } else {
                        self.store.mark_expired(
                            job.id,
                            "Cause no longer relevant in context graph",
                        ).await?;
                    }
                }
            }
        }
    }

    /// Semantic relevance check — not just "older than N minutes".
    async fn is_still_relevant(&self, job: &ReasoningJob) -> bool {
        match &job.relevance_rule {
            RelevanceRule::Always => true,
            RelevanceRule::BeforeDeadline(dl) => Utc::now() < *dl,
            RelevanceRule::ContextNodePresent { domain_key, .. } => {
                let graph = self.context_graph.read().await;
                graph.has_node(domain_key)
            }
            RelevanceRule::Custom(_) => true, // custom eval deferred
        }
    }
}
